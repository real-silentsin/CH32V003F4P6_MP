/*
 *@Note
 *GPIO routine:
 *PA0 push-pull output.
 *
 ***Only PA0--PA15 and PC16--PC17 support input pull-down.
 */

#include "debug.h"
#include "SSS_EC11Lib_V1/EC11Lib.h"

/* Global define */
#define LED1_PORT GPIOA  // PA6
#define LED1_PORT_PIN GPIO_Pin_6
#define LED1_PORT_PIN_PCC RCC_APB2Periph_GPIOA

#define LED2_PORT GPIOA  // PA7
#define LED2_PORT_PIN GPIO_Pin_7
#define LED2_PORT_PIN_PCC RCC_APB2Periph_GPIOA

/* Global Variable */
uint8_t enc_button_value = 0;

void Main_Init (void) {
    // GPIO_InitTypeDef GPIO_InitStructure = {0};

    // Активация портов для индикации (LED1-2)
    RCC_APB2PeriphClockCmd (LED1_PORT_PIN_PCC | LED2_PORT_PIN_PCC, ENABLE);

    GPIO_InitTypeDef GPIO_InitStructure1 = {LED1_PORT_PIN, GPIO_Speed_50MHz, GPIO_Mode_Out_PP};
    GPIO_Init (LED1_PORT, &GPIO_InitStructure1);

    GPIO_WriteBit (LED1_PORT, LED1_PORT_PIN, Bit_RESET);

    GPIO_InitTypeDef GPIO_InitStructure2 = {LED2_PORT_PIN, GPIO_Speed_50MHz, GPIO_Mode_Out_PP};
    GPIO_Init (LED2_PORT, &GPIO_InitStructure2);

    GPIO_WriteBit (LED2_PORT, LED2_PORT_PIN, Bit_RESET);
}

/*********************************************************************
 * @fn      main
 *
 * @brief   Main program.
 *
 * @return  none
 */
int main (void) {

    NVIC_PriorityGroupConfig (NVIC_PriorityGroup_1);
    SystemCoreClockUpdate();
    Delay_Init();

    /* USART_Printf_Init(115200);
    printf("SystemClk:%d\r\n", SystemCoreClock);
    printf( "ChipID:%08x\r\n", DBGMCU_GetCHIPID() );
    printf("EC11 TEST\r\n"); */

    Main_Init();

    EC11_Init();

    GPIO_WriteBit (LED1_PORT, LED1_PORT_PIN, Bit_SET);
    GPIO_WriteBit (LED2_PORT, LED2_PORT_PIN, Bit_SET);

    Delay_Ms (1000);

    while (1) {

        // ---------------------------------------------------------------------
        // Код для отображения состояния энкодера (для реального приложения не нужен)
        // ---------------------------------------------------------------------
        switch (EC11_GetEncoderRotationState()) {

        case ENC_EC11_TICK_CWISE:

            GPIO_WriteBit (LED1_PORT, LED1_PORT_PIN, Bit_SET);
            break;

        case ENC_EC11_TICK_CCWISE:

            GPIO_WriteBit (LED2_PORT, LED2_PORT_PIN, Bit_SET);
            break;

        default:

            GPIO_WriteBit (LED1_PORT, LED1_PORT_PIN, Bit_RESET);
            GPIO_WriteBit (LED2_PORT, LED2_PORT_PIN, Bit_RESET);
            break;
        }

        // lcd_data (' ');
        //  ---------------------------------------------------------------------
        //  Методика определения длительности нажатия кнопки энкодера
        //  ---------------------------------------------------------------------
        if (EC11_GetEncoderButtonState() != 0) {
            enc_button_value++;
        } else {

            if (enc_button_value > 0) {

                // Здесь значение времени нажатия кнопки
                // enc_button * Tpit
                GPIO_WriteBit (LED1_PORT, LED1_PORT_PIN, Bit_SET);
                GPIO_WriteBit (LED2_PORT, LED2_PORT_PIN, Bit_SET);
            }

            enc_button_value = 0;
        }
        // ---------------------------------------------------------------------
        Delay_Ms (150);
        // ---------------------------------------------------------------------
        GPIO_WriteBit (LED1_PORT, LED1_PORT_PIN, Bit_RESET);
        GPIO_WriteBit (LED2_PORT, LED2_PORT_PIN, Bit_RESET);
        // ---------------------------------------------------------------------
    }
}
